\subsection*{A3.}

\begin{enumerate}[label=(\alph*)]
\item
\begin{align*}
\E{aX_1+b} &= a\E{X_1} + b \\
           &= a\mu + b \\
           &= 0
\end{align*}
\begin{align*}
\Var{aX_1+b} &= \E{(aX_1+b)^2} - (\E{aX_1+b})^2 \\
             &= \E{a^2X_1^2 + abX_1 + b^2} \\
             &= a^2\E{X_1^2} + ab\E{X_1} + b^2 \\
             &= a^2(\Var{X_1}+(\E{X_1})^2) + ab\mu + b^2 \\
             &= a^2(\sigma^2 + \mu^2) + ab\mu + b^2 \\
             &= a^2\sigma^2 + a^2\mu^2 + ab\mu + b^2 \\
             &= a^2\sigma^2 + (a\mu + b)^2 \\
             &= a^2\sigma^2 \\
             &= 1
\end{align*}
\[
\Rightarrow a = \pm \frac{1}{\sigma}, \quad b = \mp \frac{\mu}{\sigma}
\]

\item
Let $Z = aX_1 + b \sim \mathcal{N}(0, 1)$. The pdf of $Z$: $\phi(z) = \frac{1}{\sqrt{2\pi}} e^{-z^2/2}$.
\begin{align*}
\E{\abs{aX_1 + b}} &= \E{\abs{Z}} \\
                   &= \int_{-\infty}^{\infty} \abs{z} \frac{1}{\sqrt{2\pi}} e^{-z^2/2} \dd z \\
                   &= 2 \int_{0}^{\infty} z \frac{1}{\sqrt{2\pi}} e^{-z^2/2} \dd z \\
                   &= \sqrt{\frac{2}{\pi}} \int_{0}^{\infty} z e^{-z^2/2} \dd z
\end{align*}
Let $u = \frac{z^2}{2}$ $\Rightarrow \dd u = z \dd z$:
\begin{align*}
\sqrt{\frac{2}{\pi}} \int_{0}^{\infty} e^{-u} \dd u &= \sqrt{\frac{2}{\pi}} \squarebrack{-e^{-u}}_0^\infty \\
                                                    &= \sqrt{\frac{2}{\pi}} (0 - (-1)) \\
                                                    &= \sqrt{\frac{2}{\pi}}
\end{align*}
\item
\begin{align*}
\E{X_1 + 2X_2} &= \E{X_1} + 2\E{X_2} \\
               &= \mu + 2\mu \\
               &= 3\mu \\
\Var{X_1+2X_2} &= \E{(X_1+2X_2)^2} - (\E{X_1 + 2X_2})^2 \\
               &= \E{X_1^2 + 4X_1X_2 + 4X_2^2} - (3\mu)^2 \\
               &= \E{X_1^2} + 4\E{X_1}\E{X_2} + 4\E{X_2^2} - 9\mu^2 \\
               &= (\sigma^2 + \mu^2) + 4\mu^2 +4(\sigma_2 + \mu^2) - 9\mu^2 \\
               &= 5\sigma^2
\end{align*}

\item
\begin{align*}
\E{\sqrt{n}(\hat{\mu}_n - \mu)} &= \sqrt{n}(\E{\hat{\mu}_n} - \mu) \\
    &= \sqrt{n}\left(\E{\frac{1}{n}\sum_{i=1}^{n}X_i} - \mu\right) \\
    &= \sqrt{n}\left(\frac{1}{n}\sum_{i=1}^{n}\E{X_i} - \mu\right) \\
    &= \sqrt{n}\left(\frac{1}{n} \cdot n\mu - \mu\right) \\
    &= \sqrt{n}(\mu - \mu) = 0
\end{align*}
\begin{align*}
\Var{\sqrt{n}(\hat{\mu}_n - \mu)} &= n \Var{\hat{\mu}_n} \\
    &= n \Var{\frac{1}{n}\sum_{i=1}^{n}X_i} \\
    &= n \cdot \frac{1}{n^2}\sum_{i=1}^{n}\Var{X_i} \\
    &= \frac{1}{n} \cdot n\sigma^2 = \sigma^2
\end{align*}
\end{enumerate}

\begin{enumerate}[label=(\alph*)]
\item
\begin{align*}
& \begin{bmatrix} v_1 & 0 & 0 \\ 0 & v_2 & 0 \\ 0 & 0 & v_3 \end{bmatrix}^{-1} = \begin{bmatrix} w_1 & 0 & 0 \\ 0 & w_2 & 0 \\ 0 & 0 & w_3 \end{bmatrix} \\
& \Rightarrow \begin{bmatrix} w_1 \\ w_2 \\ w_3 \end{bmatrix} = \begin{bmatrix} 1/v_1 \\ 1/v_2 \\ 1/v_3 \end{bmatrix} \\
& \Rightarrow g(v_i) = \frac{1}{v_i}
\end{align*}

\item
\begin{align*}
\|Ax\|_2^2 &= (Ax)^\top (Ax) \\
           &= x^\top A^\top A x \\
           &= x^\top I x \\
           &= \|x\|_2^2
\end{align*}

\item
Since $B$ is symmetric, $B = B^\top$. We want to show $(B^{-1})^\top = B^{-1}$.
\begin{align*}
BB^{-1} &= I \\
(BB^{-1})^\top &= I^\top \\
(B^{-1})^\top B^\top &= I \\
(B^{-1})^\top B &= I \\
(B^{-1})^\top &= B^{-1}
\end{align*}

\item
Let $\lambda$ be an eigenvalue of $C$, $\exists v \neq 0$ such that $Cv = \lambda v$.
\begin{align*}
v^\top Cv &= v^\top (\lambda v) \\
          &= \lambda v^\top v \\
          &= \lambda \|v\|_2^2
\end{align*}
Since $C$ is positive semi-definite, $v^\top Cv \geq 0$. Since $v \neq 0$, $\|v\|_2^2 > 0$.
\[
\therefore \lambda = \frac{v^\top Cv}{\|v\|_2^2} \geq 0
\]

\end{enumerate}
